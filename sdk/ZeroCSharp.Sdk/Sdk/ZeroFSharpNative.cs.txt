// The C# that bflat compiles for an F# project (Sdk.targets, ZeroFSharpNative).
// MSBuild compiles this inline task; the file isn't named .cs so that a project
// whose packages folder is inside it doesn't compile it too.
// bflat compiles only C#, and makes native code of the C# and of the
// assemblies it references, here the F# compiler's IL. This task reads that IL's
// metadata (ECMA-335 II.22-24) and writes:
//
// - for a program, a Main that calls the F# entry point: an [<EntryPoint>]
//   function, or the main@ method fsc makes for top-level code in a class C#
//   can't name. [UnsafeAccessor] reaches either by name; bflat's compiler
//   implements it, and the attributes are declared in the file (zerocore has
//   none).
// - for a native library (NativeLib=Shared), an export for each F# function
//   with [<UnmanagedCallersOnly(EntryPoint = "...")>], which calls the F# one
//   through a function pointer: bflat exports the entry points of the assembly
//   it compiles, not of the ones it references. On Windows also a .def file
//   for the import library, which names the DLL.
//
// With Interop and InteropFSharp, for a native library in C# or F#, the
// library's bindings for .NET, in C# and in F#: a [LibraryImport] (C#, on
// .NET 7 and later) or [DllImport] for each export and a copy of each struct and enum they use (WriteInterop).
//
// With Check, for every F# project, it writes nothing and checks the IL
// instead: ZERO006 for an Arena.Scratch callback whose state is a ref struct.
// C# keeps scratch memory inside the callback with `scoped`, which F# ignores;
// F#'s own rules cover the rest (a ref struct can't be returned through a
// generic result, captured, or kept in a field), but they let the callback
// store a span from the scope into a ref struct state, through which it
// outlives the callback. An input passed by value (ScratchActionIn,
// ScratchFuncIn) is a copy, so those may be ref structs.
//
// It also warns, ZERO007, where F# code calls a member C# marks `unsafe`
// (the compiler puts [RequiresUnsafe] on those) outside a function or member
// marked [<Safety "...">] or [<RequiresUnsafe>]: C# asks for an unsafe block
// there, and F# has none to ask for.
//
// System.Reflection.Metadata isn't available to inline tasks, so the tables
// are read here; only what these files need.
using System;
using System.Collections.Generic;
using System.IO;
using System.Text;
using Microsoft.Build.Framework;
using Microsoft.Build.Utilities;

public class ZeroFSharpNative : Task
{
    [Required] public string Assembly { get; set; }
    public string Source { get; set; }
    /// <summary>Check the IL (ZERO006) instead of writing C#.</summary>
    public bool Check { get; set; }
    /// <summary>For Check: the assemblies the project references, where the
    /// types its IL names from other assemblies are defined.</summary>
    public ITaskItem[] References { get; set; }
    /// <summary>For Check: whether it reported nothing. The target then notes
    /// that the assembly is checked, and skips it until it changes; one with
    /// warnings is checked, and warned about, in every build.</summary>
    [Output] public bool Clean { get; set; }
    public bool Library { get; set; }
    /// <summary>Windows libraries: where to write the .def file.</summary>
    public string DefFile { get; set; }
    /// <summary>Windows libraries: the DLL's file name, for the .def file.</summary>
    public string DllName { get; set; }
    /// <summary>A target with no command line (the C64): an entry point that
    /// takes arguments gets none, and the program has no code to read them.</summary>
    public bool NoArguments { get; set; }
    /// <summary>A native library, C# or F#: where to write its bindings for
    /// .NET, the C# that declares its exports with [LibraryImport] and copies the
    /// structs and enums they use. Written only when it changes.</summary>
    public string Interop { get; set; }
    /// <summary>The same bindings in F#, for an F# program.</summary>
    public string InteropFSharp { get; set; }
    /// <summary>For Interop: the library's name, for [DllImport] (its file
    /// name without the extension), and the class's name by default.</summary>
    public string LibraryName { get; set; }
    /// <summary>For Interop: the class the exports go in, Namespace.Class.</summary>
    public string InteropClass { get; set; }

    private byte[] _pe;
    private int _strings, _blobs, _stringIndex, _guidIndex, _blobIndex, _sections, _sectionCount;
    private readonly int[] _rows = new int[64];
    private readonly int[] _tableStart = new int[64];
    private readonly int[] _rowSize = new int[64];
    private readonly int[][] _columnOffsets = new int[64][];
    private readonly int[][] _columnSizes = new int[64][];

    private const int TypeRef = 0x01, TypeDef = 0x02, Field = 0x04, MethodDef = 0x06, Param = 0x08, MemberRef = 0x0A,
        Constant = 0x0B, ClassLayout = 0x0F, FieldLayout = 0x10,
        CustomAttribute = 0x0C, Property = 0x17, MethodSemantics = 0x18, TypeSpec = 0x1B, AssemblyTable = 0x20,
        AssemblyRef = 0x23, NestedClass = 0x29, MethodSpec = 0x2B;

    // Coded indexes (II.24.2.6): the tag bits and the tables they select.
    private static readonly int[] TypeDefOrRef = { 0x02, 0x01, 0x1B };
    private static readonly int[] HasConstant = { 0x04, 0x08, 0x17 };
    private static readonly int[] HasCustomAttribute = { 0x06, 0x04, 0x01, 0x02, 0x08, 0x09, 0x0A, 0x00, 0x0E, 0x17, 0x14,
        0x11, 0x1A, 0x1B, 0x20, 0x23, 0x26, 0x27, 0x28, 0x2A, 0x2C, 0x2B };
    private static readonly int[] HasFieldMarshal = { 0x04, 0x08 };
    private static readonly int[] HasDeclSecurity = { 0x02, 0x06, 0x20 };
    private static readonly int[] MemberRefParent = { 0x02, 0x01, 0x1A, 0x06, 0x1B };
    private static readonly int[] HasSemantics = { 0x14, 0x17 };
    private static readonly int[] MethodDefOrRef = { 0x06, 0x0A };
    private static readonly int[] MemberForwarded = { 0x04, 0x06 };
    private static readonly int[] Implementation = { 0x26, 0x23, 0x27 };
    private static readonly int[] CustomAttributeType = { -1, -1, 0x06, 0x0A, -1 };
    private static readonly int[] ResolutionScope = { 0x00, 0x1A, 0x23, 0x01 };
    private static readonly int[] TypeOrMethodDef = { 0x02, 0x06 };

        new object[] { 2, 2, TypeOrMethodDef, "s" },                     // 2A GenericParam

    private sealed class TableIndex { public int Table; }
    private static TableIndex T(int table) => new TableIndex { Table = table };

    private sealed class SignatureError : Exception
    {
        public SignatureError(string message) : base(message) { }
    }

    public override bool Execute()
    {
        _pe = File.ReadAllBytes(Assembly);
        int cli = ReadHeaders();
        if (Check)
        {
            bool ok = CheckScratchStates();
            Clean = ok;
            try { CheckUnsafeCalls(); }
            catch (Exception e)
            {
                // A warning the build can do without: IL this reader doesn't
                // follow mustn't stop it.
                Log.LogMessage(MessageImportance.Low, "ZeroCSharp.Sdk: the check for unsafe calls (ZERO007) was skipped: " + e.Message);
            }
            return ok;
        }
        if (!string.IsNullOrEmpty(Interop) || !string.IsNullOrEmpty(InteropFSharp))
        {
            WriteInterop();
            return !Log.HasLoggedErrors;
        }
        string code;
        if (Library)
        {
            code = Exports();
            if (code == null)
                return false;
        }
        else
        {
            int token = U32(cli + 20);
            if ((U32(cli + 16) & 0x10) != 0 || token >> 24 != MethodDef)
            {
                Error("The F# program has no entry point: put top-level code in its last file, or an [<EntryPoint>] function.");
                return false;
            }
            code = EntryProgram(token & 0xFFFFFF);
        }
        Directory.CreateDirectory(Path.GetDirectoryName(Path.GetFullPath(Source)));
        File.WriteAllText(Source, code);
        return true;
    }

    private void Error(string message) => Log.LogError(null, "ZERO003", null, Assembly, 0, 0, 0, 0, message);

    // ---- The program's Main ------------------------------------------------

    private string EntryProgram(int method)
    {
        string name = MethodName(method);
        int signature = SignatureStart(method);
        int parameters = _pe[signature + 1];
        bool returnsInt = _pe[signature + 2] == 0x08;
        string type = RuntimeTypeName(OwningType(method));
        // Main takes the arguments only to pass them on: a Main without
        // parameters leaves out the code that makes them into strings.
        bool takesArguments = parameters != 0 && !NoArguments;
        string call = parameters == 0 ? "FSharpMain(null)" : NoArguments ? "FSharpMain(null, new string[0])" : "FSharpMain(null, args)";
        string main = takesArguments ? "Main(string[] args)" : "Main()";
        string qualified = (type + ", " + EscapeTypeName(Path.GetFileNameWithoutExtension(Assembly))).Replace("\"", "\"\"");
        var cs = new StringBuilder();
        cs.AppendLine("// Generated by ZeroCSharp.Sdk: the native program's Main, which calls");
        cs.AppendLine("// the F# program's entry point, " + type + "::" + name + ".");
        cs.AppendLine("using System.Runtime.CompilerServices;");
        cs.AppendLine("static class ZeroFSharpEntry");
        cs.AppendLine("{");
        cs.AppendLine("    [UnsafeAccessor(UnsafeAccessorKind.StaticMethod, Name = @\"" + name.Replace("\"", "\"\"") + "\")]");
        cs.AppendLine("    static extern " + (returnsInt ? "int" : "void") + " FSharpMain([UnsafeAccessorType(@\"" + qualified
            + "\")] object type" + (parameters == 0 ? "" : ", string[] args") + ");");
        cs.AppendLine(returnsInt
            ? "    static int " + main + " => " + call + ";"
            : "    static int " + main + " { " + call + "; return 0; }");
        cs.AppendLine("}");
        cs.AppendLine("namespace System.Runtime.CompilerServices");
        cs.AppendLine("{");
        cs.AppendLine("    internal enum UnsafeAccessorKind { Constructor, Method, StaticMethod, Field, StaticField }");
        cs.AppendLine("    [AttributeUsage(AttributeTargets.Method)]");
        cs.AppendLine("    internal sealed class UnsafeAccessorAttribute : Attribute");
        cs.AppendLine("    {");
        cs.AppendLine("        public UnsafeAccessorAttribute(UnsafeAccessorKind kind) { Kind = kind; }");
        cs.AppendLine("        public UnsafeAccessorKind Kind { get; }");
        cs.AppendLine("        public string Name { get; set; }");
        cs.AppendLine("    }");
        cs.AppendLine("    [AttributeUsage(AttributeTargets.Parameter | AttributeTargets.ReturnValue)]");
        cs.AppendLine("    internal sealed class UnsafeAccessorTypeAttribute : Attribute");
        cs.AppendLine("    {");
        cs.AppendLine("        public UnsafeAccessorTypeAttribute(string typeName) { TypeName = typeName; }");
        cs.AppendLine("        public string TypeName { get; }");
        cs.AppendLine("    }");
        cs.AppendLine("}");
        return cs.ToString();
    }

    // ---- A native library's exports ----------------------------------------

    // The methods with [UnmanagedCallersOnly(EntryPoint = ...)]: entry point, MethodDef row.
    private List<KeyValuePair<string, int>> FindExports()
    {
        var exports = new List<KeyValuePair<string, int>>();
        for (int row = 1; row <= _rows[CustomAttribute]; row++)
        {
            int parent = Cell(CustomAttribute, row, 0);
            if (parent % 32 != 0) continue;    // tag 0: a MethodDef
            if (!IsUnmanagedCallersOnly(Cell(CustomAttribute, row, 1))) continue;
            string entryPoint = NamedStringArgument(Cell(CustomAttribute, row, 2), "EntryPoint");
            if (entryPoint != null)
                exports.Add(new KeyValuePair<string, int>(entryPoint, parent >> 5));
        }
        return exports;
    }

    private string Exports()
    {
        var exports = FindExports();
        var cs = new StringBuilder();
        cs.AppendLine("// Generated by ZeroCSharp.Sdk: the native library's exports, the F#");
        cs.AppendLine("// functions with [<UnmanagedCallersOnly(EntryPoint = ...)>], each called");
        cs.AppendLine("// through a function pointer (bflat exports what it compiles from C#).");
        cs.AppendLine("static unsafe class ZeroFSharpExports");
        cs.AppendLine("{");
        bool ok = true;
        int n = 0;
        foreach (var export in exports)
        {
            int method = export.Value;
            string target = CSharpTypeName(OwningType(method), out string why);
            if (target == null || (MethodFlags(method) & 0x7) != 6)
            {
                Log.LogError(null, "ZERO005", null, Assembly, 0, 0, 0, 0,
                    "The export " + export.Key + " (" + MethodName(method) + ") isn't public"
                    + (why != null ? " (" + why + ")" : "") + ": exported functions and the modules around them must be public.");
                ok = false;
                continue;
            }
            List<string> types;
            try { types = ExportSignature(method); }
            catch (SignatureError e)
            {
                Log.LogError(null, "ZERO005", null, Assembly, 0, 0, 0, 0,
                    "The export " + export.Key + " (" + MethodName(method) + ") " + e.Message);
                ok = false;
                continue;
            }
            string ret = types[0];
            var parameters = new StringBuilder();
            var arguments = new StringBuilder();
            for (int i = 1; i < types.Count; i++)
            {
                if (i > 1) { parameters.Append(", "); arguments.Append(", "); }
                parameters.Append(types[i]).Append(" a").Append(i);
                arguments.Append('a').Append(i);
            }
            var pointer = new StringBuilder("delegate* unmanaged<");
            for (int i = 1; i < types.Count; i++) pointer.Append(types[i]).Append(", ");
            pointer.Append(ret).Append('>');
            cs.AppendLine("    [System.Runtime.InteropServices.UnmanagedCallersOnly(EntryPoint = @\"" + export.Key.Replace("\"", "\"\"") + "\")]");
            cs.AppendLine("    static " + ret + " Export" + n++ + "(" + parameters + ") => ((" + pointer + ")&"
                + target + "." + MethodName(method) + ")(" + arguments + ");");
        }
        cs.AppendLine("}");
        if (!ok) return null;
        if (!string.IsNullOrEmpty(DefFile))
        {
            var def = new StringBuilder();
            def.AppendLine("LIBRARY \"" + DllName + "\"");
            def.AppendLine("EXPORTS");
            foreach (var export in exports) def.AppendLine("    " + export.Key);
            Directory.CreateDirectory(Path.GetDirectoryName(Path.GetFullPath(DefFile)));
            File.WriteAllText(DefFile, def.ToString());
        }
        return cs.ToString();
    }

    // ---- Check: Arena.Scratch states (ZERO006) ------------------------------

    private bool CheckScratchStates()
    {
        bool ok = true;
        for (int row = 1; row <= _rows[TypeSpec]; row++)
        {
            int p = _blobs + Cell(TypeSpec, row, 0);
            CompressedLength(p, out p);
            // GENERICINST CLASS ScratchAction`1<TState> or ScratchFunc`2<TState, TResult>
            if (_pe[p] != 0x15 || _pe[p + 1] != 0x12) continue;
            p += 2;
            int coded = CompressedInt(ref p);
            if (TypeDefOrRef[coded % 4] != TypeRef) continue;
            int delegateRow = coded >> 2;
            string name = Str(Cell(TypeRef, delegateRow, 1));
            if ((name != "ScratchAction`1" && name != "ScratchFunc`2") || Str(Cell(TypeRef, delegateRow, 2)) != ""
                || ReferencedAssembly(delegateRow) != "BflatLib")
                continue;
            CompressedInt(ref p);    // the argument count
            int state = p;
            string refStruct = RefStructName(ref p);
            if (refStruct == null) continue;
            refStruct = refStruct.Split('`')[0];
            string where = StateUser(state, p - state);
            Log.LogError(null, "ZERO006", null, Assembly, 0, 0, 0, 0,
                "An Arena.Scratch callback" + (where != null ? " (" + where + ")" : "") + " has a ref struct, " + refStruct
                + ", as its state. F# doesn't check what the callback stores in it, so a span from the scope could outlive"
                + " the callback. Pass it by value instead, Arena.Scratch(input, ScratchFuncIn<...>(fun scratch input -> ...)):"
                + " the callback then gets a copy. Or use a state that isn't a ref struct.");
            ok = false;
        }
        return ok;
    }

    // The name of the ref struct at signature p (moved past the type), or null.
    private string RefStructName(ref int p)
    {
        int element = _pe[p];
        if (element == 0x11 || (element == 0x15 && _pe[p + 1] == 0x11))
        {
            bool generic = element == 0x15;
            p += generic ? 2 : 1;
            int coded = CompressedInt(ref p);
            if (generic)
            {
                int count = CompressedInt(ref p);
                for (int i = 0; i < count; i++) SkipType(ref p);
            }
            int table = TypeDefOrRef[coded % 4], row = coded >> 2;
            if (table == TypeDef)
                return HasAttribute(row, "IsByRefLikeAttribute") ? Str(Cell(TypeDef, row, 1)) : null;
            if (table == TypeRef)
            {
                string name = Str(Cell(TypeRef, row, 1));
                return IsRefStructRef(row) ? name : null;
            }
            return null;
        }
        SkipType(ref p);
        return null;
    }

    // A type another assembly defines: whether it's a ref struct, read from
    // that assembly (one of References, found by its assembly name).
    private bool IsRefStructRef(int row)
    {
        var path = new List<string>();
        int r = row;
        while (true)
        {
            path.Insert(0, Str(Cell(TypeRef, r, 1)));
            int scope = Cell(TypeRef, r, 0);
            if (ResolutionScope[scope % 4] != TypeRef) break;
            r = scope >> 2;
        }
        string ns = Str(Cell(TypeRef, r, 2));
        ZeroFSharpNative other = ReferenceNamed(ReferencedAssembly(row));
        if (other == null) return false;
        int type = other.FindType(ns, path);
        return type != 0 && other.HasAttribute(type, "IsByRefLikeAttribute");
    }

    private readonly Dictionary<string, ZeroFSharpNative> _references = new Dictionary<string, ZeroFSharpNative>();

    private ZeroFSharpNative ReferenceNamed(string assemblyName)
    {
        if (assemblyName == null || References == null) return null;
        if (_references.Count == 0)
        {
            foreach (ITaskItem item in References)
            {
                string file = item.GetMetadata("FullPath");
                if (!File.Exists(file)) continue;
                var reference = new ZeroFSharpNative { Assembly = file };
                try
                {
                    reference._pe = File.ReadAllBytes(file);
                    reference.ReadHeaders();
                }
                catch (Exception) { continue; }    // not an assembly this reader handles
                if (reference._rows[AssemblyTable] == 0) continue;
                string name = reference.Str(reference.Cell(AssemblyTable, 1, 7));
                if (!_references.ContainsKey(name)) _references.Add(name, reference);
            }
        }
        _references.TryGetValue(assemblyName, out ZeroFSharpNative found);
        return found;
    }

    // The assembly a TypeRef (or the type it's nested in) comes from, or null.
    private string ReferencedAssembly(int row)
    {
        while (true)
        {
            int scope = Cell(TypeRef, row, 0);
            int table = ResolutionScope[scope % 4];
            if (table == AssemblyRef) return Str(Cell(AssemblyRef, scope >> 2, 6));
            if (table != TypeRef) return null;
            row = scope >> 2;
        }
    }

    // The TypeDef named namespace + path (outer type first), or 0.
    private int FindType(string ns, List<string> path)
    {
        int found = 0;
        for (int i = 0; i < path.Count; i++)
        {
            int match = 0;
            for (int t = 1; t <= _rows[TypeDef] && match == 0; t++)
            {
                if (Str(Cell(TypeDef, t, 1)) != path[i] || EnclosingType(t) != found) continue;
                if (i == 0 && Str(Cell(TypeDef, t, 2)) != ns) continue;
                match = t;
            }
            if (match == 0) return 0;
            found = match;
        }
        return found;
    }

    // Whether the TypeDef has an attribute of this name (any namespace).
    private bool HasAttribute(int type, string attribute) => HasAttribute(3, type, attribute);

    // The same for a row of another table, by its HasCustomAttribute tag:
    // 0 MethodDef, 3 TypeDef, 9 Property.
    private bool HasAttribute(int tag, int row, string attribute) => AttributeParents(attribute).Contains(row << 5 | tag);

    private readonly Dictionary<string, HashSet<int>> _attributeParents = new Dictionary<string, HashSet<int>>();

    // Everything that has an attribute of this name, as HasCustomAttribute values.
    private HashSet<int> AttributeParents(string attribute)
    {
        if (_attributeParents.TryGetValue(attribute, out HashSet<int> parents)) return parents;
        parents = new HashSet<int>();
        for (int row = 1; row <= _rows[CustomAttribute]; row++)
        {
            int typeToken = AttributeType(Cell(CustomAttribute, row, 1));
            int table = typeToken >> 24, typeRow = typeToken & 0xFFFFFF;
            if ((table == TypeRef || table == TypeDef) && Str(Cell(table, typeRow, 1)) == attribute)
                parents.Add(Cell(CustomAttribute, row, 0));
        }
        _attributeParents.Add(attribute, parents);
        return parents;
    }

    // ---- Check: calls to unsafe members (ZERO007) -----------------------------

    private int[] _methodOwner;          // MethodDef row -> TypeDef row
    private int[] _fieldOwner;           // Field row -> TypeDef row
    private int[] _enclosing;            // TypeDef row -> the TypeDef it's nested in, or 0
    private Dictionary<int, int> _propertyOfAccessor;

    // Each call in a method's IL to a member with [RequiresUnsafe] is reported
    // unless the method is acknowledged: it, or its property, has [<Safety>]
    // or [<RequiresUnsafe>], or it is code fsc made for an acknowledged
    // method. That code is a closure class or a lifted inner function, with
    // '@' in its name; it belongs to the methods whose IL mentions it (which
    // create the closure or call the function).
    //
    // [<Safety "why">] is FSharp.Core's NoCompilerInlining under another name
    // (fsharp/core/attrs.fs), so that fsc leaves the marked function in one
    // piece: a small one copied into its callers would put the unsafe call
    // in methods nobody marked. It reaches the IL as NoCompilerInlining with
    // the reason, which is how it's told from the plain attribute.
    private void CheckUnsafeCalls()
    {
        int methods = _rows[MethodDef], types = _rows[TypeDef];
        _methodOwner = Owners(5, methods);
        _fieldOwner = Owners(4, _rows[0x04]);
        _enclosing = new int[types + 1];
        for (int row = 1; row <= _rows[NestedClass]; row++)
            _enclosing[Cell(NestedClass, row, 0)] = Cell(NestedClass, row, 1);
        var acknowledged = new bool[methods + 1];
        var generatedMethod = new bool[methods + 1];
        var generatedType = new bool[types + 1];
        for (int t = 1; t <= types; t++)
        {
            int enclosing = _enclosing[t];    // an enclosing type has the smaller row
            string name = Str(Cell(TypeDef, t, 1));
            generatedType[t] = name.IndexOf('@') >= 0 || (enclosing != 0 && generatedType[enclosing]);
        }
        HashSet<int> safety = SafetyParents();
        for (int m = 1; m <= methods; m++)
        {
            generatedMethod[m] = generatedType[_methodOwner[m]] || MethodName(m).IndexOf('@') >= 0;
            acknowledged[m] = IsMarked(m, safety) || IsMarked(m, AttributeParents("RequiresUnsafeAttribute"));
        }

        // What each method's IL mentions: unsafe members it calls, and the
        // generated types and methods that are its own code.
        var unsafeCalls = new List<KeyValuePair<string, int>>[methods + 1];    // the member, and the call's IL offset
        var usersOfType = new List<int>[types + 1];
        var usersOfMethod = new List<int>[methods + 1];
        var unsafeRefs = new Dictionary<int, string>();    // MemberRef row -> the member's name, or null
        for (int m = 1; m <= methods; m++)
        {
            int rva = Cell(MethodDef, m, 0);
            if (rva == 0) continue;
            int p = FileOffset(rva), end;
            if ((_pe[p] & 3) == 2) { end = p + 1 + (_pe[p] >> 2); p++; }
            else { int size = U32(p + 4); p += (U16(p) >> 12) * 4; end = p + size; }
            int code = p;
            while (p < end)
            {
                int offset = p - code;
                int op = _pe[p++];
                if (op == 0xFE) op = 0xFE00 | _pe[p++];
                int operand = OperandSize(op, p);
                if (!HasToken(op)) { p += operand; continue; }
                int token = U32(p);
                p += 4;
                bool call = op == 0x28 || op == 0x6F || op == 0x73 || op == 0x27 || op == 0xFE06 || op == 0xFE07;
                int table = (int)((uint)token >> 24), row = token & 0xFFFFFF;
                if (table == MethodSpec)
                {
                    int coded = Cell(MethodSpec, row, 0);
                    table = MethodDefOrRef[coded % 2];
                    row = coded >> 1;
                }
                if (table == MethodDef)
                {
                    Uses(usersOfType, generatedType, _methodOwner[row], m);
                    if (generatedMethod[row] && row != m) Add(usersOfMethod, row, m);
                    if (call && (IsMarked(row, "RequiresUnsafeAttribute")))
                        Add(unsafeCalls, m, Call(DisplayName(RuntimeTypeName(_methodOwner[row]).Replace("+", "."), MethodName(row)), offset));
                }
                else if (table == 0x04)
                    Uses(usersOfType, generatedType, _fieldOwner[row], m);
                else if (table == MemberRef)
                {
                    int parent = Cell(MemberRef, row, 0);
                    int parentTable = MemberRefParent[parent % 8], parentRow = parent >> 3;
                    if (parentTable == TypeSpec)
                    {
                        // GENERICINST (CLASS | VALUETYPE) type ...: the generic type's member
                        int q = _blobs + Cell(TypeSpec, parentRow, 0);
                        CompressedLength(q, out q);
                        if (_pe[q] != 0x15) continue;
                        q += 2;
                        int coded = CompressedInt(ref q);
                        parentTable = TypeDefOrRef[coded % 4];
                        parentRow = coded >> 2;
                    }
                    if (parentTable == TypeDef)
                    {
                        Uses(usersOfType, generatedType, parentRow, m);
                        if (call && UnsafeMethodNamed(parentRow, Str(Cell(MemberRef, row, 1)), Shape(_blobs + Cell(MemberRef, row, 2))))
                            Add(unsafeCalls, m, Call(DisplayName(RuntimeTypeName(parentRow).Replace("+", "."), Str(Cell(MemberRef, row, 1))), offset));
                    }
                    else if (parentTable == TypeRef && call)
                    {
                        if (!unsafeRefs.TryGetValue(row, out string member))
                        {
                            member = UnsafeMemberRef(parentRow, row);
                            unsafeRefs.Add(row, member);
                        }
                        if (member != null) Add(unsafeCalls, m, Call(member, offset));
                    }
                }
            }
        }

        // Generated code is acknowledged with the code it belongs to.
        for (bool changed = true; changed;)
        {
            changed = false;
            for (int m = 1; m <= methods; m++)
            {
                if (acknowledged[m] || !generatedMethod[m]) continue;
                bool covered = AnyAcknowledged(usersOfMethod[m], acknowledged);
                for (int t = _methodOwner[m]; t != 0 && !covered; t = _enclosing[t])
                    covered = generatedType[t] && AnyAcknowledged(usersOfType[t], acknowledged);
                if (covered) { acknowledged[m] = true; changed = true; }
            }
        }

        var reported = new HashSet<string>();
        Pdb pdb = null;
        for (int m = 1; m <= methods; m++)
        {
            if (unsafeCalls[m] == null || acknowledged[m]) continue;
            if (pdb == null) pdb = Pdb.Read(Path.ChangeExtension(Assembly, ".pdb"));
            // A lambda's call is reported for the function it's written in,
            // which is the one to mark.
            int function = m;
            for (int step = 0; step < 32 && generatedMethod[function]; step++)
            {
                List<int> users = usersOfMethod[function];
                for (int t = _methodOwner[function]; t != 0 && users == null; t = _enclosing[t])
                    if (generatedType[t]) users = usersOfType[t];
                if (users == null) break;
                function = users[0];
            }
            foreach (KeyValuePair<string, int> call in unsafeCalls[m])
            {
                string member = call.Key, file = Assembly;
                int line = 0, column = 0;
                bool located = pdb != null && pdb.Locate(m, call.Value, out file, out line, out column);
                if (!located) file = Assembly;
                // One warning for a line, or without lines for a function.
                if (!reported.Add(function + " " + member + " " + line)) continue;
                string where = function == m ? "" : " (in a lambda or an inner function" + (located ? "" : LineOf(m)) + ")";
                Clean = false;
                Log.LogWarning(null, "ZERO007", null, file, line, column, 0, 0,
                    CallerName(function) + " calls " + member + where + ", which is unsafe: the compiler can't check that it's used correctly,"
                    + " and C# asks for an unsafe block around it. Read the member's safety rules, then mark the function"
                    + " [<Safety \"why this use is correct\">], or [<RequiresUnsafe>] to make it unsafe to call in turn.");
            }
        }
    }

    private static KeyValuePair<string, int> Call(string member, int offset) => new KeyValuePair<string, int>(member, offset);

    private static void Add<T>(List<T>[] lists, int index, T item)
    {
        if (lists[index] == null) lists[index] = new List<T>();
        lists[index].Add(item);
    }

    // Method user mentions a member of type: recorded for generated types,
    // unless it is the type's own code.
    private void Uses(List<int>[] usersOfType, bool[] generatedType, int type, int user)
    {
        if (type == 0 || !generatedType[type]) return;
        for (int t = _methodOwner[user]; t != 0; t = _enclosing[t])
            if (t == type) return;
        Add(usersOfType, type, user);
    }

    private static bool AnyAcknowledged(List<int> users, bool[] acknowledged)
    {
        if (users != null)
            foreach (int user in users)
                if (acknowledged[user]) return true;
        return false;
    }

    // What has [<Safety "why">]: NoCompilerInliningAttribute, made with a
    // reason that isn't empty (II.23.3: the prolog, then the string).
    private HashSet<int> SafetyParents()
    {
        var parents = new HashSet<int>();
        for (int row = 1; row <= _rows[CustomAttribute]; row++)
        {
            int typeToken = AttributeType(Cell(CustomAttribute, row, 1));
            int table = typeToken >> 24, typeRow = typeToken & 0xFFFFFF;
            // On ZeroTarget=dotnet, FSharp.Core is .NET's, whose NoCompilerInlining
            // takes no argument: [<Safety>] is an attribute of its own there
            // (dotnet/ZeroLib.FSharp/Attributes.fs), with the reason likewise.
            if (table != TypeRef && table != TypeDef) continue;
            string attributeName = Str(Cell(table, typeRow, 1));
            if (attributeName != "NoCompilerInliningAttribute" && attributeName != "SafetyAttribute") continue;
            int length = CompressedLength(_blobs + Cell(CustomAttribute, row, 2), out int p);
            if (length < 5 || U16(p) != 1 || _pe[p + 2] == 0xFF) continue;    // no argument, or null
            p += 2;
            string reason = SerString(ref p);
            if (reason != null && reason.Trim().Length > 0) parents.Add(Cell(CustomAttribute, row, 0));
        }
        return parents;
    }

    private bool IsMarked(int method, string attribute) => IsMarked(method, AttributeParents(attribute));

    // Whether the method, or the property it's an accessor of, is among the
    // marked (HasCustomAttribute values: tag 0 MethodDef, 9 Property).
    private bool IsMarked(int method, HashSet<int> marked)
    {
        if (marked.Contains(method << 5)) return true;
        if (_propertyOfAccessor == null)
        {
            _propertyOfAccessor = new Dictionary<int, int>();
            for (int row = 1; row <= _rows[MethodSemantics]; row++)
            {
                int association = Cell(MethodSemantics, row, 2);
                if ((association & 1) == 1) _propertyOfAccessor[Cell(MethodSemantics, row, 1)] = association >> 1;
            }
        }
        return _propertyOfAccessor.TryGetValue(method, out int property) && marked.Contains(property << 5 | 9);
    }

    // For each row of a table whose rows TypeDefs own in runs (Field: TypeDef
    // column 4, MethodDef: column 5), the TypeDef that owns it.
    private int[] Owners(int column, int rows)
    {
        var owners = new int[rows + 1];
        int types = _rows[TypeDef];
        for (int t = 1; t <= types; t++)
        {
            int first = Cell(TypeDef, t, column), next = t < types ? Cell(TypeDef, t + 1, column) : rows + 1;
            for (int row = first; row < next && row <= rows; row++) owners[row] = t;
        }
        return owners;
    }

    // Whether the type has a method of this name and parameter count with
    // [RequiresUnsafe] (on it or its property).
    private bool UnsafeMethodNamed(int type, string name, string shape)
    {
        if (_methodOwner == null) _methodOwner = Owners(5, _rows[MethodDef]);
        int first = Cell(TypeDef, type, 5), next = type < _rows[TypeDef] ? Cell(TypeDef, type + 1, 5) : _rows[MethodDef] + 1;
        for (int m = first; m < next; m++)
            if (MethodName(m) == name && Shape(_blobs + Cell(MethodDef, m, 4)) == shape
                && IsMarked(m, "RequiresUnsafeAttribute"))
                return true;
        return false;
    }

    // A member of a type another assembly defines: Type.Member if that
    // assembly marks it unsafe, or null.
    private string UnsafeMemberRef(int typeRef, int memberRef)
    {
        var path = new List<string>();
        int r = typeRef;
        while (true)
        {
            path.Insert(0, Str(Cell(TypeRef, r, 1)));
            int scope = Cell(TypeRef, r, 0);
            if (ResolutionScope[scope % 4] != TypeRef) break;
            r = scope >> 2;
        }
        ZeroFSharpNative other = ReferenceNamed(ReferencedAssembly(typeRef));
        if (other == null) return null;
        int type = other.FindType(Str(Cell(TypeRef, r, 2)), path);
        string name = Str(Cell(MemberRef, memberRef, 1));
        if (type == 0 || !other.UnsafeMethodNamed(type, name, Shape(_blobs + Cell(MemberRef, memberRef, 2))))
            return null;
        return DisplayName(path[path.Count - 1], name);
    }

    // Arena.Reset, List.AsSpan, Game.Peak (for get_Peak).
    private static string DisplayName(string type, string member)
    {
        if (member.StartsWith("get_") || member.StartsWith("set_")) member = member.Substring(4);
        return type.Split('`')[0] + "." + member;
    }

    // The F# function for the message. Top-level code is main@ or a static
    // constructor in <StartupCode$assembly>.$File.
    private string CallerName(int method)
    {
        int type = _methodOwner[method];
        if (Str(Cell(TypeDef, type, 2)).StartsWith("<StartupCode$"))
            return "The top-level code of " + Str(Cell(TypeDef, type, 1)).TrimStart('$');
        return DisplayName(RuntimeTypeName(type).Replace("+", "."), MethodName(method));
    }

    // ", line 8" for code fsc named xs@8 or clo@8-1 (a closure class, or a
    // lifted function), or "".
    private string LineOf(int method)
    {
        string name = MethodName(method);
        for (int t = _methodOwner[method]; t != 0 && name.IndexOf('@') < 0; t = _enclosing[t])
            name = Str(Cell(TypeDef, t, 1));
        int at = name.IndexOf('@'), end = at + 1;
        if (at < 0) return "";
        while (end < name.Length && char.IsDigit(name[end])) end++;
        return end > at + 1 ? ", line " + name.Substring(at + 1, end - at - 1) : "";
    }

    // A method signature's parameter count (II.23.2.1, 23.2.2).
    private int ParameterCount(int blob)
    {
        CompressedLength(blob, out int p);
        int convention = _pe[p++];
        if ((convention & 0x10) != 0) CompressedInt(ref p);    // the generic parameter count
        return CompressedInt(ref p);
    }

    // Whether the instruction's operand is a metadata token of a member or a
    // type: the calls, the field instructions, ldtoken.
    private static bool HasToken(int op)
    {
        switch (op)
        {
            case 0x27: case 0x28: case 0x6F: case 0x73:                           // jmp, call, callvirt, newobj
            case 0x7B: case 0x7C: case 0x7D: case 0x7E: case 0x7F: case 0x80:     // the field instructions
            case 0xD0: case 0xFE06: case 0xFE07:                                  // ldtoken, ldftn, ldvirtftn
                return true;
        }
        return false;
    }

    // The size of an instruction's operand (III.1.2.1, and each instruction's
    // description); p is where the operand starts.
    private int OperandSize(int op, int p)
    {
        if (op >= 0xFE00)
        {
            switch (op & 0xFF)
            {
                case 0x06: case 0x07: case 0x15: case 0x16: case 0x1C: return 4;    // ldftn, ldvirtftn, initobj, constrained., sizeof
                case 0x09: case 0x0A: case 0x0B: case 0x0C: case 0x0D: case 0x0E: return 2;    // ldarg .. stloc
                case 0x12: case 0x19: return 1;    // unaligned., no.
                default: return 0;
            }
        }
        if (op == 0x45) return 4 + 4 * U32(p);    // switch
        if (op == 0x21 || op == 0x23) return 8;   // ldc.i8, ldc.r8
        if ((op >= 0x0E && op <= 0x13) || op == 0x1F || (op >= 0x2B && op <= 0x37) || op == 0xDE) return 1;
        if (op == 0x20 || op == 0x22 || (op >= 0x27 && op <= 0x29) || (op >= 0x38 && op <= 0x44)
            || (op >= 0x6F && op <= 0x75) || op == 0x79 || (op >= 0x7B && op <= 0x81) || op == 0x8C || op == 0x8D
            || op == 0x8F || (op >= 0xA3 && op <= 0xA5) || op == 0xC2 || op == 0xC6 || op == 0xD0 || op == 0xDD)
            return 4;
        return 0;
    }

    // The method that takes the state (the callback F# made of the lambda),
    // as Type.method, for the message; null if none is found. The state's
    // signature is the bytes [start, start + length) of the blob heap; the
    // callback takes it by reference: BYREF, then the same bytes.
    private string StateUser(int start, int length)
    {
        for (int method = 1; method <= _rows[MethodDef]; method++)
        {
            int p = _blobs + Cell(MethodDef, method, 4);
            int size = CompressedLength(p, out p);
            for (int i = p; i + length < p + size; i++)
            {
                if (_pe[i] != 0x10) continue;
                bool same = true;
                for (int j = 0; j < length && same; j++) same = _pe[i + 1 + j] == _pe[start + j];
                if (same) return RuntimeTypeName(OwningType(method)).Replace("+", ".") + "." + MethodName(method);
            }
        }
        return null;
    }

    // A method signature's shape, to tell overloads apart from another
    // assembly, whose tokens differ: the generic parameter count, then the
    // return type and each parameter's, with classes and structs by name.
    // Two overloads with as many parameters (Arena.Scratch's) differ here.
    private string Shape(int blob)
    {
        CompressedLength(blob, out int p);
        int convention = _pe[p++];
        var shape = new System.Text.StringBuilder();
        if ((convention & 0x10) != 0) shape.Append('g').Append(CompressedInt(ref p));
        int parameters = CompressedInt(ref p);
        for (int i = 0; i <= parameters; i++)
        {
            shape.Append(i == 0 ? "(" : ",");
            TypeShape(ref p, shape);
        }
        return shape.Append(')').ToString();
    }

    private void TypeShape(ref int p, System.Text.StringBuilder shape)
    {
        int start = p, element = _pe[p];
        switch (element)
        {
            case 0x11: case 0x12:    // VALUETYPE, CLASS
                p++;
                shape.Append(TypeDefOrRefName(CompressedInt(ref p)));
                return;
            case 0x13: case 0x1E:    // VAR, MVAR
                p++;
                shape.Append(element == 0x13 ? "!" : "!!").Append(CompressedInt(ref p));
                return;
            case 0x0F: case 0x10: case 0x1D: case 0x45:    // PTR, BYREF, SZARRAY, PINNED
                p++;
                shape.Append(element.ToString("x2")).Append(' ');
                TypeShape(ref p, shape);
                return;
            case 0x1F: case 0x20:    // a custom modifier: not part of the shape
                p++;
                CompressedInt(ref p);
                TypeShape(ref p, shape);
                return;
            case 0x15:    // GENERICINST: the type, then its arguments
                p++;
                shape.Append('<');
                TypeShape(ref p, shape);
                int count = CompressedInt(ref p);
                for (int i = 0; i < count; i++)
                {
                    shape.Append(' ');
                    TypeShape(ref p, shape);
                }
                shape.Append('>');
                return;
            default:    // a primitive, or an array or function pointer: its kind
                SkipType(ref p);
                shape.Append(element.ToString("x2"));
                return;
        }
    }

    // The name (namespace and name) of a TypeDefOrRef coded index's type;
    // "?" for a TypeSpec, which a signature doesn't name this way.
    private string TypeDefOrRefName(int coded)
    {
        int tag = coded & 3, row = coded >> 2;
        if (tag == 0) return Str(Cell(TypeDef, row, 2)) + "." + Str(Cell(TypeDef, row, 1));
        if (tag == 1) return Str(Cell(TypeRef, row, 2)) + "." + Str(Cell(TypeRef, row, 1));
        return "?";
    }

    // Moves p past a type in a signature (II.23.2.12).
    private void SkipType(ref int p)
    {
        int element = _pe[p++];
        switch (element)
        {
            case 0x11: case 0x12:    // VALUETYPE, CLASS
                CompressedInt(ref p);
                break;
            case 0x13: case 0x1E:    // VAR, MVAR
                CompressedInt(ref p);
                break;
            case 0x0F: case 0x10: case 0x1D: case 0x45:    // PTR, BYREF, SZARRAY, PINNED
                SkipType(ref p);
                break;
            case 0x1F: case 0x20:    // a custom modifier, then the type
                CompressedInt(ref p);
                SkipType(ref p);
                break;
            case 0x15:    // GENERICINST
                p++;
                CompressedInt(ref p);
                int count = CompressedInt(ref p);
                for (int i = 0; i < count; i++) SkipType(ref p);
                break;
            case 0x14:    // ARRAY: type, rank, sizes, lower bounds
                SkipType(ref p);
                CompressedInt(ref p);
                int sizes = CompressedInt(ref p);
                for (int i = 0; i < sizes; i++) CompressedInt(ref p);
                int bounds = CompressedInt(ref p);
                for (int i = 0; i < bounds; i++) CompressedInt(ref p);
                break;
            case 0x1B:    // FNPTR: a method signature
                p++;
                int parameters = CompressedInt(ref p);
                for (int i = 0; i <= parameters; i++) SkipType(ref p);
                break;
            default:    // a primitive type: one byte
                break;
        }
    }

    private int AttributeType(int attributeType)
    {
        int table = CustomAttributeType[attributeType % 8], row = attributeType >> 3;
        if (table == MemberRef)
        {
            int parent = Cell(MemberRef, row, 0);
            return (MemberRefParent[parent % 8] << 24) | (parent >> 3);
        }
        if (table == MethodDef)
            return (TypeDef << 24) | OwningType(row);
        return 0;
    }

    private bool IsUnmanagedCallersOnly(int attributeType)
    {
        int table = CustomAttributeType[attributeType % 8], row = attributeType >> 3;
        int typeToken;
        if (table == MemberRef)
        {
            int parent = Cell(MemberRef, row, 0);
            typeToken = (MemberRefParent[parent % 8] << 24) | (parent >> 3);
        }
        else if (table == MethodDef)
            typeToken = (TypeDef << 24) | OwningType(row);
        else
            return false;
        int typeTable = typeToken >> 24, typeRow = typeToken & 0xFFFFFF;
        if (typeTable != TypeRef && typeTable != TypeDef) return false;
        int nameColumn = typeTable == TypeRef ? 1 : 1, namespaceColumn = typeTable == TypeRef ? 2 : 2;
        return Str(Cell(typeTable, typeRow, nameColumn)) == "UnmanagedCallersOnlyAttribute"
            && Str(Cell(typeTable, typeRow, namespaceColumn)) == "System.Runtime.InteropServices";
    }

    // A custom attribute's named string argument (II.23.3), or null.
    private string NamedStringArgument(int blob, string wanted)
    {
        int length = CompressedLength(_blobs + blob, out int p);
        int end = p + length;
        if (U16(p) != 1) return null;
        p += 2;    // UnmanagedCallersOnlyAttribute's constructor has no arguments
        int count = U16(p);
        p += 2;
        for (int i = 0; i < count && p < end; i++)
        {
            p++;    // FIELD (0x53) or PROPERTY (0x54)
            int type = _pe[p++];
            if (type == 0x1D) type = 0x1D00 | _pe[p++];    // SZARRAY of
            string name = SerString(ref p);
            switch (type)
            {
                case 0x0E:    // string
                    string value = SerString(ref p);
                    if (name == wanted) return value;
                    break;
                case 0x1D50:    // System.Type[], CallConvs
                    int items = (int)(uint)U32(p);
                    p += 4;
                    if (items != -1)
                        for (int j = 0; j < items; j++) SerString(ref p);
                    break;
                default:
                    return null;
            }
        }
        return null;
    }

    private string SerString(ref int p)
    {
        if (_pe[p] == 0xFF) { p++; return null; }
        int length = CompressedLength(p, out int start);
        p = start + length;
        return Encoding.UTF8.GetString(_pe, start, length);
    }

    // The C# types of an export's return value and parameters.
    private List<string> ExportSignature(int method)
    {
        int p = _blobs + Cell(MethodDef, method, 4);
        CompressedLength(p, out p);
        int convention = _pe[p++];
        if ((convention & 0x10) != 0) throw new SignatureError("is generic; an export can't be.");
        if ((convention & 0x20) != 0) throw new SignatureError("isn't static.");
        int count = CompressedInt(ref p);
        var types = new List<string>();
        for (int i = 0; i <= count; i++)
            types.Add(SignatureType(ref p, i == 0));
        return types;
    }

    private string SignatureType(ref int p, bool isReturn)
    {
        int element = _pe[p++];
        switch (element)
        {
            case 0x01: if (isReturn) return "void"; break;
            case 0x02: throw new SignatureError("takes or returns a bool, which isn't blittable: use byte or int.");
            case 0x03: throw new SignatureError("takes or returns a char, which isn't blittable: use uint16.");
            case 0x04: return "sbyte";
            case 0x05: return "byte";
            case 0x06: return "short";
            case 0x07: return "ushort";
            case 0x08: return "int";
            case 0x09: return "uint";
            case 0x0A: return "long";
            case 0x0B: return "ulong";
            case 0x0C: return "float";
            case 0x0D: return "double";
            case 0x18: return "nint";
            case 0x19: return "nuint";
            case 0x0F:    // PTR
                if (_pe[p] == 0x01) { p++; return "void*"; }
                return SignatureType(ref p, false) + "*";
            case 0x11:    // VALUETYPE
                int coded = CompressedInt(ref p);
                int table = TypeDefOrRef[coded % 4], row = coded >> 2;
                string name = table == TypeDef ? CSharpTypeName(row, out _) : table == TypeRef ? CSharpTypeRefName(row) : null;
                if (name == null) throw new SignatureError("uses a struct C# can't name (a non-public or generic one).");
                return name;
            case 0x1F:
            case 0x20:    // a custom modifier, then the type
                CompressedInt(ref p);
                return SignatureType(ref p, isReturn);
        }
        throw new SignatureError("takes or returns a type an export can't: use numbers, pointers, nativeint and structs of them.");
    }

    // ---- A native library's bindings for .NET -------------------------------

    // The code a .NET program compiles to call the library, in C# (Interop)
    // and F# (InteropFSharp): a [LibraryImport] or [DllImport] for each export, under its
    // method's name, and a copy of each struct and enum those use, with the
    // same name, fields and layout. The IL is read once into BType and BDef;
    // each language prints them. An export they can't declare (a type from
    // another assembly, say) is left out with a warning, ZERO008: the library
    // itself is fine.

    // A type in a signature or a field.
    private sealed class BType
    {
        public string CSharp, FSharp;    // a primitive's names
        public BType Pointee;            // a pointer: what it points at (null: void*)
        public bool IsPointer, IsVoid;
        public BDef Struct;              // one of the library's structs or enums
        public string Convention;        // a function pointer: C#'s calling convention
        public List<BType> Signature;    // a function pointer: parameters, then the return type
    }

    // One of the library's structs or enums.
    private sealed class BDef
    {
        public string Namespace, Name;
        public bool IsEnum, IsExplicit;
        public BType Underlying;                     // an enum's
        public int Element;                          // an enum's element type, for F#'s literals
        public List<KeyValuePair<string, string>> Members = new List<KeyValuePair<string, string>>();
        public int Pack, Size;
        public List<BField> Fields = new List<BField>();
    }

    private sealed class BField
    {
        public string Name;
        public BType Type;
        public int Offset = -1;      // explicit layout
        public int Count;            // a fixed buffer: Type is its element
        public int BufferSize;
    }

    private sealed class Binding
    {
        public string EntryPoint, Name, Key;
        public BType Return;
        public List<BType> Types = new List<BType>();
        public List<string> Names = new List<string>();
    }

    private readonly Dictionary<int, BDef> _bindingDefs = new Dictionary<int, BDef>();
    private readonly Dictionary<int, string> _bindingErrors = new Dictionary<int, string>();
    private readonly List<BDef> _bindingOrder = new List<BDef>();    // each after the ones its fields use

    private void WriteInterop()
    {
        string library = string.IsNullOrEmpty(LibraryName) ? Path.GetFileNameWithoutExtension(Assembly) : LibraryName;
        string fullClass = string.IsNullOrEmpty(InteropClass) ? library : InteropClass;
        var parts = new List<string>();
        foreach (string part in fullClass.Split('.')) parts.Add(Identifier(part, "_"));
        string className = parts[parts.Count - 1];
        string ns = parts.Count > 1 ? string.Join(".", parts.GetRange(0, parts.Count - 1)) : "";

        var bindings = new List<Binding>();
        foreach (var export in FindExports())
        {
            int method = export.Value;
            var b = new Binding { EntryPoint = export.Key };
            try
            {
                int p = _blobs + Cell(MethodDef, method, 4);
                CompressedLength(p, out p);
                int convention = _pe[p++];
                if ((convention & 0x30) != 0) continue;    // generic or an instance method: ILC refuses it
                int count = CompressedInt(ref p);
                b.Return = BindingType(ref p, true, false);
                for (int i = 0; i < count; i++) b.Types.Add(BindingType(ref p, false, false));
            }
            catch (SignatureError e)
            {
                Log.LogWarning(null, "ZERO008", null, Assembly, 0, 0, 0, 0,
                    "The export " + export.Key + " (" + MethodName(method) + ") isn't in the .NET bindings: it " + e.Message);
                continue;
            }
            string[] names = ParameterNames(method, b.Types.Count);
            var taken = new HashSet<string>();
            for (int i = 0; i < b.Types.Count; i++)
            {
                string name = names[i] != null && IsIdentifier(names[i]) ? names[i] : "arg" + (i + 1);
                while (!taken.Add(name)) name += "_";
                b.Names.Add(name);
            }
            string methodName = MethodName(method);
            b.Name = IsIdentifier(methodName) ? methodName : Identifier(export.Key, "Export");
            bindings.Add(b);
        }
        // Two exports with the same name and parameters (Add in two classes,
        // say) take their entry points' names instead.
        var keys = new Dictionary<string, int>();
        foreach (var b in bindings)
        {
            var key = new StringBuilder(b.Name).Append('(');
            foreach (var t in b.Types) key.Append(CSharpType(t)).Append(',');
            b.Key = key.ToString();
            keys[b.Key] = keys.TryGetValue(b.Key, out int n) ? n + 1 : 1;
        }
        foreach (var b in bindings)
            if (keys[b.Key] > 1 || b.Name == className || b.Name == "LibraryName") b.Name = Identifier(b.EntryPoint, "Export");

        if (!string.IsNullOrEmpty(Interop)) WriteIfChanged(Interop, CSharpBindings(library, ns, className, bindings));
        if (!string.IsNullOrEmpty(InteropFSharp)) WriteIfChanged(InteropFSharp, FSharpBindings(library, ns, className, bindings));
    }

    private static void WriteIfChanged(string file, string text)
    {
        string path = Path.GetFullPath(file);
        if (File.Exists(path) && File.ReadAllText(path) == text) return;
        Directory.CreateDirectory(Path.GetDirectoryName(path));
        File.WriteAllText(path, text);
    }

    // A parameter's type. pointee: behind a pointer, where bool and char can
    // be (they're one and two bytes on both sides).
    private BType BindingType(ref int p, bool isReturn, bool pointee)
    {
        int element = _pe[p++];
        switch (element)
        {
            case 0x01: if (isReturn || pointee) return new BType { IsVoid = true, CSharp = "void", FSharp = "void" }; break;
            case 0x02: if (pointee) return Primitive("bool", "bool"); throw new SignatureError("takes or returns a bool, which .NET marshals as four bytes: use byte or int.");
            case 0x03: if (pointee) return Primitive("char", "char"); throw new SignatureError("takes or returns a char, which .NET marshals as one byte: use ushort.");
            case 0x04: return Primitive("sbyte", "sbyte");
            case 0x05: return Primitive("byte", "byte");
            case 0x06: return Primitive("short", "int16");
            case 0x07: return Primitive("ushort", "uint16");
            case 0x08: return Primitive("int", "int");
            case 0x09: return Primitive("uint", "uint32");
            case 0x0A: return Primitive("long", "int64");
            case 0x0B: return Primitive("ulong", "uint64");
            case 0x0C: return Primitive("float", "float32");
            case 0x0D: return Primitive("double", "float");
            case 0x18: return Primitive("nint", "nativeint");
            case 0x19: return Primitive("nuint", "unativeint");
            case 0x0F:    // PTR
            {
                BType pointee2 = BindingType(ref p, false, true);
                return new BType { IsPointer = true, Pointee = pointee2.IsVoid ? null : pointee2 };
            }
            case 0x11:    // VALUETYPE
            {
                int coded = CompressedInt(ref p);
                int table = TypeDefOrRef[coded % 4], row = coded >> 2;
                if (table == TypeRef)
                    throw new SignatureError("uses " + Str(Cell(TypeRef, row, 1)) + ", a struct from another assembly: the bindings copy only the library's own structs.");
                if (table != TypeDef) break;
                return new BType { Struct = BindingStruct(row) };
            }
            case 0x1B:    // FNPTR: a method signature
            {
                int convention = _pe[p++];
                string callingConvention;
                switch (convention & 0x0F)
                {
                    case 0x01: callingConvention = "unmanaged[Cdecl]"; break;
                    case 0x02: callingConvention = "unmanaged[Stdcall]"; break;
                    case 0x03: callingConvention = "unmanaged[Thiscall]"; break;
                    case 0x04: callingConvention = "unmanaged[Fastcall]"; break;
                    case 0x09: callingConvention = "unmanaged"; break;
                    default: throw new SignatureError("takes or returns a managed function pointer: use delegate* unmanaged.");
                }
                int count = CompressedInt(ref p);
                BType ret = BindingType(ref p, true, false);
                var signature = new List<BType>();
                for (int i = 0; i < count; i++) signature.Add(BindingType(ref p, false, false));
                signature.Add(ret);
                return new BType { Convention = callingConvention, Signature = signature };
            }
            case 0x1F:
            case 0x20:    // a custom modifier, then the type
                CompressedInt(ref p);
                return BindingType(ref p, isReturn, pointee);
        }
        throw new SignatureError("takes or returns a type that can't cross to .NET: use numbers, pointers, nint and structs of them.");
    }

    private static BType Primitive(string csharp, string fsharp) => new BType { CSharp = csharp, FSharp = fsharp };

    // A struct or enum of the library's, read once (a struct that points at
    // itself is fine).
    private BDef BindingStruct(int type)
    {
        if (_bindingDefs.TryGetValue(type, out BDef known)) return known;
        if (_bindingErrors.TryGetValue(type, out string error)) throw new SignatureError(error);
        string name = Str(Cell(TypeDef, type, 1)), ns = Str(Cell(TypeDef, type, 2));
        var def = new BDef { Namespace = ns, Name = name };
        try
        {
            if (EnclosingType(type) != 0)
                throw new SignatureError("uses " + name + ", a struct nested in another type: the bindings copy only top-level structs.");
            if (!IsIdentifier(name))
                throw new SignatureError("uses " + name + ", which isn't a C# name.");
            foreach (string part in ns.Split('.'))
                if (ns.Length > 0 && !IsIdentifier(part)) throw new SignatureError("uses " + name + ", in " + ns + ", which isn't a C# namespace.");
            _bindingDefs[type] = def;
            if (BaseTypeName(type) == "System.Enum") ReadEnum(type, def);
            else ReadStruct(type, def);
            _bindingOrder.Add(def);
            return def;
        }
        catch (SignatureError e)
        {
            _bindingDefs.Remove(type);
            _bindingErrors[type] = e.Message;
            throw;
        }
    }

    private void ReadStruct(int type, BDef def)
    {
        int flags = Cell(TypeDef, type, 0), layout = flags & 0x18;
        if (layout == 0)
            throw new SignatureError("uses " + def.Name + ", a struct with automatic layout: .NET may order its fields differently.");
        for (int row = 1; row <= _rows[ClassLayout]; row++)
            if (Cell(ClassLayout, row, 2) == type) { def.Pack = Cell(ClassLayout, row, 0); def.Size = Cell(ClassLayout, row, 1); }
        def.IsExplicit = layout == 0x10;
        int number = 0;
        foreach (int field in Fields(type))
        {
            if ((Cell(Field, field, 0) & 0x10) != 0) continue;    // static
            number++;
            var f = new BField { Name = FieldName(Str(Cell(Field, field, 1)), number) };
            if (def.IsExplicit)
                for (int row = 1; row <= _rows[FieldLayout]; row++)
                    if (Cell(FieldLayout, row, 1) == field) f.Offset = Cell(FieldLayout, row, 0);
            int p = _blobs + Cell(Field, field, 2);
            CompressedLength(p, out p);
            p++;    // FIELD
            while (_pe[p] == 0x1F || _pe[p] == 0x20) { p++; CompressedInt(ref p); }
            if (!FixedBuffer(p, f)) f.Type = BindingType(ref p, false, false);
            def.Fields.Add(f);
        }
    }

    // A C# fixed-size buffer: the field's type is a nested <Name>e__FixedBuffer
    // struct with the buffer's size and one field of its element type.
    private bool FixedBuffer(int p, BField f)
    {
        if (_pe[p] != 0x11) return false;
        p++;
        int coded = CompressedInt(ref p);
        if (TypeDefOrRef[coded % 4] != TypeDef) return false;
        int type = coded >> 2;
        if (EnclosingType(type) == 0 || !Str(Cell(TypeDef, type, 1)).EndsWith("e__FixedBuffer", StringComparison.Ordinal)) return false;
        int size = 0;
        for (int row = 1; row <= _rows[ClassLayout]; row++)
            if (Cell(ClassLayout, row, 2) == type) size = Cell(ClassLayout, row, 1);
        foreach (int field in Fields(type))
        {
            int q = _blobs + Cell(Field, field, 2);
            CompressedLength(q, out q);
            int element = _pe[q + 1];
            int width = element == 0x02 || element == 0x04 || element == 0x05 ? 1 : element == 0x03 || element == 0x06 || element == 0x07 ? 2
                : element == 0x08 || element == 0x09 || element == 0x0C ? 4 : element == 0x0A || element == 0x0B || element == 0x0D ? 8 : 0;
            if (width == 0 || size == 0) return false;
            q++;
            f.Type = BindingType(ref q, false, true);
            f.Count = size / width;
            f.BufferSize = size;
            return true;
        }
        return false;
    }

    private void ReadEnum(int type, BDef def)
    {
        def.IsEnum = true;
        foreach (int field in Fields(type))
        {
            int flags = Cell(Field, field, 0);
            int p = _blobs + Cell(Field, field, 2);
            CompressedLength(p, out p);
            p++;    // FIELD
            if ((flags & 0x10) == 0) { def.Element = _pe[p]; def.Underlying = BindingType(ref p, false, false); continue; }    // value__
            if ((flags & 0x40) == 0) continue;    // not a literal
            for (int row = 1; row <= _rows[Constant]; row++)
            {
                int parent = Cell(Constant, row, 1);
                if (parent % 4 != 0 || parent >> 2 != field) continue;    // tag 0: a Field
                int value = _blobs + Cell(Constant, row, 2);
                CompressedLength(value, out value);
                def.Members.Add(new KeyValuePair<string, string>(Str(Cell(Field, field, 1)),
                    ConstantText(_pe[_tableStart[Constant] + (row - 1) * _rowSize[Constant]], value)));
            }
        }
        if (def.Underlying == null) throw new SignatureError("uses " + def.Name + ", an enum without a value field.");
    }

    // An enum constant in decimal.
    private string ConstantText(int element, int p)
    {
        switch (element)
        {
            case 0x04: return ((sbyte)_pe[p]).ToString();
            case 0x05: return _pe[p].ToString();
            case 0x06: return ((short)U16(p)).ToString();
            case 0x07: return U16(p).ToString();
            case 0x08: return U32(p).ToString();
            case 0x09: return ((uint)U32(p)).ToString();
            case 0x0A: return ((long)(uint)U32(p) | (long)U32(p + 4) << 32).ToString();
            case 0x0B: return ((ulong)(uint)U32(p) | (ulong)(uint)U32(p + 4) << 32).ToString();
            default: throw new SignatureError("has an enum constant of an unknown type.");
        }
    }

    // ---- The bindings in C# ------------------------------------------------

    private static string CSharpType(BType t)
    {
        if (t.IsPointer) return (t.Pointee == null ? "void" : CSharpType(t.Pointee)) + "*";
        if (t.Struct != null) return "global::" + (t.Struct.Namespace.Length > 0 ? t.Struct.Namespace + "." : "") + t.Struct.Name;
        if (t.Signature != null)
        {
            var s = new StringBuilder("delegate* " + t.Convention + "<");
            for (int i = 0; i < t.Signature.Count; i++) s.Append(i > 0 ? ", " : "").Append(CSharpType(t.Signature[i]));
            return s.Append('>').ToString();
        }
        return t.CSharp;
    }

    private static bool CSharpUnsafe(BType t) => t.IsPointer || t.Signature != null;

    private string CSharpBindings(string library, string ns, string className, List<Binding> bindings)
    {
        bool anyUnsafe = false;
        foreach (var b in bindings)
        {
            anyUnsafe |= CSharpUnsafe(b.Return);
            foreach (var t in b.Types) anyUnsafe |= CSharpUnsafe(t);
        }
        var cls = new StringBuilder();
        // @: the [LibraryImport] generator copies the name into a file of its
        // own, which the #pragma below doesn't cover, and a lower-case name
        // there (a library called mathlib) would warn, CS8981.
        cls.Append("public static " + (anyUnsafe ? "unsafe " : "") + "partial class @" + className + "\n{\n");
        cls.Append("    public const string LibraryName = @\"" + library.Replace("\"", "\"\"") + "\";\n");
        foreach (var b in bindings)
        {
            // [LibraryImport] where the program's .NET has it (7 and later):
            // its source generator writes the call at compile time, which
            // NativeAOT and trimming prefer. [DllImport] before that, or when
            // the program defines ZERO_INTEROP_DLLIMPORT.
            string entryPoint = "EntryPoint = @\"" + b.EntryPoint.Replace("\"", "\"\"") + "\"";
            var parameters = new StringBuilder();
            for (int i = 0; i < b.Types.Count; i++)
                parameters.Append(i > 0 ? ", " : "").Append(CSharpType(b.Types[i])).Append(' ').Append(Escape(b.Names[i]));
            string signature = CSharpType(b.Return) + " " + Escape(b.Name) + "(" + parameters + ");\n";
            cls.Append('\n');
            cls.Append("#if NET7_0_OR_GREATER && !ZERO_INTEROP_DLLIMPORT\n");
            cls.Append("    [global::System.Runtime.InteropServices.LibraryImport(LibraryName, " + entryPoint + ")]\n");
            cls.Append("    public static partial " + signature);
            cls.Append("#else\n");
            cls.Append("    [global::System.Runtime.InteropServices.DllImport(LibraryName, " + entryPoint + ", ExactSpelling = true)]\n");
            cls.Append("    public static extern " + signature);
            cls.Append("#endif\n");
        }
        cls.Append("}\n");
        // The class, then the types, a block for each namespace.
        var definitions = new List<KeyValuePair<string, string>> { new KeyValuePair<string, string>(ns, cls.ToString()) };
        foreach (var def in _bindingOrder) definitions.Add(new KeyValuePair<string, string>(def.Namespace, CSharpDefinition(def)));
        var namespaces = new List<string>();
        foreach (var d in definitions) if (!namespaces.Contains(d.Key)) namespaces.Add(d.Key);
        var cs = new StringBuilder();
        cs.Append("// <auto-generated/>\n");
        cs.Append("// The .NET bindings of the native library " + library + ", made by ZeroCSharp.Sdk\n");
        cs.Append("// from its IL: its exports, and the structs and enums they use. Regenerated\n");
        cs.Append("// by each build of the library; don't edit it.\n");
        cs.Append("#pragma warning disable\n");
        foreach (string space in namespaces)
        {
            var block = new StringBuilder();
            foreach (var d in definitions)
                if (d.Key == space) block.Append(block.Length > 0 ? "\n" : "").Append(d.Value);
            cs.Append('\n');
            cs.Append(space.Length > 0 ? "namespace " + space + "\n{\n" + Indent(block.ToString()) + "}\n" : block.ToString());
        }
        return cs.ToString();
    }

    private static string CSharpDefinition(BDef def)
    {
        if (def.IsEnum)
        {
            var members = new StringBuilder();
            foreach (var m in def.Members) members.Append("    ").Append(Escape(m.Key)).Append(" = ").Append(m.Value).Append(",\n");
            return "public enum " + def.Name + " : " + def.Underlying.CSharp + "\n{\n" + members + "}\n";
        }
        var body = new StringBuilder();
        bool isUnsafe = false;
        foreach (var f in def.Fields)
        {
            if (f.Offset >= 0) body.Append("    [global::System.Runtime.InteropServices.FieldOffset(" + f.Offset + ")]\n");
            if (f.Count > 0)
            {
                isUnsafe = true;
                body.Append("    public fixed ").Append(CSharpType(f.Type)).Append(' ').Append(Escape(f.Name)).Append('[').Append(f.Count).Append("];\n");
                continue;
            }
            isUnsafe |= CSharpUnsafe(f.Type);
            body.Append("    public ").Append(CSharpType(f.Type)).Append(' ').Append(Escape(f.Name)).Append(";\n");
        }
        var arguments = new List<string> { "global::System.Runtime.InteropServices.LayoutKind." + (def.IsExplicit ? "Explicit" : "Sequential") };
        if (def.Pack != 0) arguments.Add("Pack = " + def.Pack);
        if (def.Size != 0) arguments.Add("Size = " + def.Size);
        var cs = new StringBuilder();
        if (def.IsExplicit || arguments.Count > 1)
            cs.Append("[global::System.Runtime.InteropServices.StructLayout(" + string.Join(", ", arguments) + ")]\n");
        cs.Append("public " + (isUnsafe ? "unsafe " : "") + "struct " + def.Name + "\n{\n").Append(body).Append("}\n");
        return cs.ToString();
    }

    // ---- The bindings in F# ------------------------------------------------
    // The same, for an F# program, which can't compile C#: a module of
    // extern functions, [<Struct>] types with val fields, and enums. F# has no
    // fixed buffers or function-pointer types: a fixed buffer is a struct of
    // the buffer's size (Name_FieldBuffer, its first element as Element0),
    // and a function pointer a nativeint.

    private static string FSharpType(BType t)
    {
        if (t.IsPointer)
        {
            if (t.Pointee == null) return "voidptr";
            string inner = FSharpType(t.Pointee);
            return "nativeptr<" + inner + ">";
        }
        if (t.Struct != null) return (t.Struct.Namespace.Length > 0 ? "global." + FSharpPath(t.Struct.Namespace) + "." : "") + FSharpName(t.Struct.Name);
        if (t.Signature != null) return "nativeint";
        return t.FSharp;
    }

    // In an extern declaration F# takes C's pointer syntax, not nativeptr<_>.
    private static string FSharpExternType(BType t)
    {
        if (t.IsPointer) return (t.Pointee == null ? "void" : FSharpExternType(t.Pointee)) + "*";
        return FSharpType(t);
    }

    private static string FSharpPath(string ns)
    {
        var parts = ns.Split('.');
        for (int i = 0; i < parts.Length; i++) parts[i] = FSharpName(parts[i]);
        return string.Join(".", parts);
    }

    private string FSharpBindings(string library, string ns, string className, List<Binding> bindings)
    {
        var fs = new StringBuilder();
        fs.Append("// <auto-generated/>\n");
        fs.Append("// The .NET bindings of the native library " + library + ", made by ZeroCSharp.Sdk\n");
        fs.Append("// from its IL: its exports, and the structs and enums they use. Regenerated\n");
        fs.Append("// by each build of the library; don't edit it. Compile it before the files\n");
        fs.Append("// that use it.\n");
        fs.Append("#nowarn \"9\" \"51\" \"3391\"\n");
        // The types in the order their fields need them, a namespace block
        // (rec, for a struct that points at itself) each time it changes.
        string current = null;
        foreach (var def in _bindingOrder)
        {
            if (def.Namespace != current)
            {
                current = def.Namespace;
                fs.Append('\n').Append("namespace rec ").Append(current.Length > 0 ? FSharpPath(current) : "global").Append("\n\n");
                fs.Append("open System.Runtime.InteropServices\n");
            }
            fs.Append('\n').Append(FSharpDefinition(def));
        }
        fs.Append('\n').Append("namespace ").Append(ns.Length > 0 ? FSharpPath(ns) : "global").Append("\n\n");
        fs.Append("open System.Runtime.InteropServices\n\n");
        fs.Append("module " + FSharpName(className) + " =\n");
        fs.Append("    [<Literal>]\n");
        fs.Append("    let LibraryName = \"" + library.Replace("\\", "\\\\").Replace("\"", "\\\"") + "\"\n");
        foreach (var b in bindings)
        {
            fs.Append('\n');
            fs.Append("    [<DllImport(LibraryName, EntryPoint = \"" + b.EntryPoint.Replace("\\", "\\\\").Replace("\"", "\\\"") + "\", ExactSpelling = true)>]\n");
            var parameters = new StringBuilder();
            for (int i = 0; i < b.Types.Count; i++)
                parameters.Append(i > 0 ? ", " : "").Append(FSharpExternType(b.Types[i])).Append(' ').Append(FSharpName(b.Names[i]));
            fs.Append("    extern " + FSharpExternType(b.Return) + " " + FSharpName(b.Name) + "(" + parameters + ")\n");
        }
        return fs.ToString();
    }

    private static string FSharpDefinition(BDef def)
    {
        var fs = new StringBuilder();
        if (def.IsEnum)
        {
            string suffix = FSharpLiteralSuffix(def.Element);
            fs.Append("type " + FSharpName(def.Name) + " =\n");
            if (def.Members.Count == 0) fs.Append("    | None = 0" + suffix + "\n");
            foreach (var m in def.Members) fs.Append("    | ").Append(FSharpName(m.Key)).Append(" = ").Append(m.Value).Append(suffix).Append('\n');
            return fs.ToString();
        }
        // A fixed buffer's struct, before the struct that holds it.
        foreach (var f in def.Fields)
        {
            if (f.Count == 0) continue;
            fs.Append("[<Struct; StructLayout(LayoutKind.Sequential, Size = " + f.BufferSize + ")>]\n");
            fs.Append("type " + FSharpName(def.Name + "_" + f.Name + "Buffer") + " =\n");
            fs.Append("    val mutable Element0: " + FSharpType(f.Type) + "\n\n");
        }
        var arguments = new List<string> { "LayoutKind." + (def.IsExplicit ? "Explicit" : "Sequential") };
        if (def.Pack != 0) arguments.Add("Pack = " + def.Pack);
        if (def.Size != 0) arguments.Add("Size = " + def.Size);
        fs.Append("[<Struct; StructLayout(" + string.Join(", ", arguments) + ")>]\n");
        fs.Append("type " + FSharpName(def.Name) + " =\n");
        if (def.Fields.Count == 0) fs.Append("    struct end\n");
        foreach (var f in def.Fields)
        {
            string type = f.Count > 0 ? FSharpName(def.Name + "_" + f.Name + "Buffer") : FSharpType(f.Type);
            fs.Append("    ");
            if (f.Offset >= 0) fs.Append("[<FieldOffset(" + f.Offset + ")>] ");
            // F# doesn't count a pointer as zero-initializable: without this
            // the struct's default constructor, Node(), would be refused.
            if (f.Count == 0 && f.Type.IsPointer) fs.Append("[<DefaultValue(false)>] ");
            fs.Append("val mutable ").Append(FSharpName(f.Name)).Append(": ").Append(type).Append('\n');
        }
        return fs.ToString();
    }

    private static string FSharpLiteralSuffix(int element)
    {
        switch (element)
        {
            case 0x04: return "y";
            case 0x05: return "uy";
            case 0x06: return "s";
            case 0x07: return "us";
            case 0x09: return "u";
            case 0x0A: return "L";
            case 0x0B: return "UL";
            default: return "";
        }
    }

    private static readonly HashSet<string> FSharpKeywords = new HashSet<string>(("abstract and as assert base begin class default delegate do done downcast downto elif else end exception extern false finally fixed for fun function global if in inherit inline interface internal lazy let match member module mutable namespace new not null of open or override private public rec return select sig static struct then to true try type upcast use val void when while with yield const break checked component constraint continue event external include mixin parallel process protected pure sealed tailcall trait virtual").Split(' '));

    private static string FSharpName(string name) => FSharpKeywords.Contains(name) ? "``" + name + "``" : name;

    // ---- Helpers for the bindings ------------------------------------------

    // The type's fields: from its FieldList to the next type's.
    private IEnumerable<int> Fields(int type)
    {
        int first = Cell(TypeDef, type, 4);
        int end = type < _rows[TypeDef] ? Cell(TypeDef, type + 1, 4) : _rows[Field] + 1;
        for (int field = first; field < end; field++) yield return field;
    }

    private string BaseTypeName(int type)
    {
        int coded = Cell(TypeDef, type, 3);
        int table = TypeDefOrRef[coded % 4], row = coded >> 2;
        if (row == 0) return null;
        if (table == TypeRef) return Str(Cell(TypeRef, row, 2)) + "." + Str(Cell(TypeRef, row, 1));
        if (table == TypeDef) return Str(Cell(TypeDef, row, 2)) + "." + Str(Cell(TypeDef, row, 1));
        return null;
    }

    // A field's name: an auto-property's backing field <X>k__BackingField
    // takes the property's.
    private static string FieldName(string name, int number)
    {
        if (name.StartsWith("<", StringComparison.Ordinal))
        {
            int end = name.IndexOf(">k__BackingField", StringComparison.Ordinal);
            if (end > 1 && IsIdentifier(name.Substring(1, end - 1))) return name.Substring(1, end - 1);
        }
        return IsIdentifier(name) ? name : "field" + number;
    }

    // The names of a method's parameters, from its Param rows (sequence 0 is
    // the return value); null where there's none.
    private string[] ParameterNames(int method, int count)
    {
        var names = new string[count];
        int first = Cell(MethodDef, method, 5);
        int end = method < _rows[MethodDef] ? Cell(MethodDef, method + 1, 5) : _rows[Param] + 1;
        for (int row = first; row < end; row++)
        {
            int sequence = Cell(Param, row, 1);
            if (sequence >= 1 && sequence <= count) names[sequence - 1] = Str(Cell(Param, row, 2));
        }
        return names;
    }

    // Any text as a C# identifier: other characters become _.
    private static string Identifier(string text, string fallback)
    {
        var b = new StringBuilder();
        foreach (char c in text) b.Append(char.IsLetterOrDigit(c) || c == '_' ? c : '_');
        if (b.Length == 0) return fallback;
        if (char.IsDigit(b[0])) b.Insert(0, '_');
        return b.ToString();
    }

    private static readonly HashSet<string> Keywords = new HashSet<string>(("abstract as base bool break byte case catch char checked class const continue decimal default delegate do double else enum event explicit extern false finally fixed float for foreach goto if implicit in int interface internal is lock long namespace new null object operator out override params private protected public readonly ref return sbyte sealed short sizeof stackalloc static string struct switch this throw true try typeof uint ulong unchecked unsafe ushort using virtual void volatile while").Split(' '));

    private static string Escape(string name) => Keywords.Contains(name) ? "@" + name : name;

    private static string Indent(string text)
    {
        var b = new StringBuilder();
        foreach (string line in text.TrimEnd('\n').Split('\n')) b.Append(line.Length > 0 ? "    " + line : line).Append('\n');
        return b.ToString();
    }

    // ---- Types and methods --------------------------------------------------

    private string MethodName(int method) => Str(Cell(MethodDef, method, 3));
    private int MethodFlags(int method) => Cell(MethodDef, method, 2);

    private int SignatureStart(int method)
    {
        CompressedLength(_blobs + Cell(MethodDef, method, 4), out int start);
        return start;
    }

    // The TypeDef whose method list holds the method.
    private int OwningType(int method)
    {
        int type = 0;
        for (int t = 1; t <= _rows[TypeDef]; t++)
            if (Cell(TypeDef, t, 5) <= method) type = t;
        return type;
    }

    private int EnclosingType(int type)
    {
        for (int row = 1; row <= _rows[NestedClass]; row++)
            if (Cell(NestedClass, row, 0) == type) return Cell(NestedClass, row, 1);
        return 0;
    }

    // Namespace.Outer+Inner, escaped as type names are (for UnsafeAccessorType).
    private string RuntimeTypeName(int type)
    {
        int enclosing = EnclosingType(type);
        string name = EscapeTypeName(Str(Cell(TypeDef, type, 1)));
        if (enclosing != 0) return RuntimeTypeName(enclosing) + "+" + name;
        string ns = Str(Cell(TypeDef, type, 2));
        return ns.Length > 0 ? EscapeTypeName(ns) + "." + name : name;
    }

    // global::Namespace.Outer.Inner, or null for a type C# can't name or reach.
    private string CSharpTypeName(int type, out string why)
    {
        why = null;
        int flags = Cell(TypeDef, type, 0), visibility = flags & 0x7;
        string name = Str(Cell(TypeDef, type, 1));
        if (visibility != 1 && visibility != 2) { why = name + " isn't public"; return null; }
        if (!IsIdentifier(name)) { why = name + " isn't a C# name"; return null; }
        int enclosing = EnclosingType(type);
        if (enclosing != 0)
        {
            string outer = CSharpTypeName(enclosing, out why);
            return outer == null ? null : outer + "." + name;
        }
        string ns = Str(Cell(TypeDef, type, 2));
        foreach (string part in ns.Split('.'))
            if (ns.Length > 0 && !IsIdentifier(part)) { why = ns + " isn't a C# namespace"; return null; }
        return "global::" + (ns.Length > 0 ? ns + "." : "") + name;
    }

    private string CSharpTypeRefName(int row)
    {
        string name = Str(Cell(TypeRef, row, 1)), ns = Str(Cell(TypeRef, row, 2));
        int scope = Cell(TypeRef, row, 0);
        if (ResolutionScope[scope % 4] == TypeRef)
        {
            string outer = CSharpTypeRefName(scope >> 2);
            return outer == null ? null : outer + "." + name;
        }
        if (!IsIdentifier(name)) return null;
        return "global::" + (ns.Length > 0 ? ns + "." : "") + name;
    }

    private static bool IsIdentifier(string name)
    {
        if (name.Length == 0 || !(char.IsLetter(name[0]) || name[0] == '_')) return false;
        foreach (char c in name)
            if (!(char.IsLetterOrDigit(c) || c == '_')) return false;
        return true;
    }

    // Type names escape , + & * [ ] \ with a backslash.
    private static string EscapeTypeName(string name)
    {
        var b = new StringBuilder();
        foreach (char c in name)
        {
            if (",+&*[]\\".IndexOf(c) >= 0) b.Append('\\');
            b.Append(c);
        }
        return b.ToString();
    }

    // ---- Source lines, from the portable PDB --------------------------------

    // The PDB the F# compiler wrote next to the assembly, for the file and
    // line of a call: its Document and MethodDebugInformation tables (the
    // Portable PDB format, in dotnet/runtime's docs/design/specs). Without one
    // (DebugType none or embedded), warnings name the function only.
    private sealed class Pdb
    {
        private byte[] _data;
        private int _blobs, _blobIndex, _documentIndex, _documents, _documentRow, _methods, _methodRows;

        public static Pdb Read(string path)
        {
            var pdb = new Pdb();
            try
            {
                if (!File.Exists(path) || !pdb.ReadTables(File.ReadAllBytes(path))) pdb._methodRows = 0;
            }
            catch (Exception) { pdb._methodRows = 0; }
            return pdb;
        }

        private int U16(int o) => _data[o] | _data[o + 1] << 8;
        private int U32(int o) => _data[o] | _data[o + 1] << 8 | _data[o + 2] << 16 | _data[o + 3] << 24;

        private bool ReadTables(byte[] data)
        {
            _data = data;
            if (U32(0) != 0x424A5342) return false;    // "BSJB": not a portable PDB (an old Windows one)
            int streamList = 16 + U32(12), streamCount = U16(streamList + 2), cursor = streamList + 4, tables = 0;
            for (int i = 0; i < streamCount; i++)
            {
                int offset = U32(cursor), nameStart = cursor + 8, nameEnd = nameStart;
                while (_data[nameEnd] != 0) nameEnd++;
                string name = Encoding.ASCII.GetString(_data, nameStart, nameEnd - nameStart);
                if (name == "#~") tables = offset;
                else if (name == "#Blob") _blobs = offset;
                cursor = nameStart + ((nameEnd - nameStart + 4) & ~3);
            }
            if (tables == 0) return false;
            int heapSizes = _data[tables + 6];
            int guidIndex = (heapSizes & 2) != 0 ? 4 : 2;
            _blobIndex = (heapSizes & 4) != 0 ? 4 : 2;
            long valid = (long)(uint)U32(tables + 8) | (long)(uint)U32(tables + 12) << 32;
            int counts = tables + 24, documentRows = 0;
            for (int t = 0; t < 64; t++)
            {
                if ((valid >> t & 1) == 0) continue;
                if (t < 0x30) return false;    // a table before Document: not the layout read here
                if (t == 0x30) documentRows = U32(counts);
                if (t == 0x31) _methodRows = U32(counts);
                counts += 4;
            }
            // Document: Name (blob), HashAlgorithm (guid), Hash (blob), Language (guid).
            // MethodDebugInformation: Document, SequencePoints (blob).
            _documentIndex = documentRows > 0xFFFF ? 4 : 2;
            _documentRow = 2 * _blobIndex + 2 * guidIndex;
            _documents = counts;
            _methods = counts + documentRows * _documentRow;
            return true;
        }

        private int Index(int o, int size) => size == 4 ? U32(o) : U16(o);

        private int Compressed(ref int p)
        {
            int b = _data[p];
            if ((b & 0x80) == 0) { p += 1; return b; }
            if ((b & 0xC0) == 0x80) { p += 2; return (b & 0x3F) << 8 | _data[p - 1]; }
            p += 4;
            return (b & 0x1F) << 24 | _data[p - 3] << 16 | _data[p - 2] << 8 | _data[p - 1];
        }

        // A signed compressed integer (II.23.2): the sign is the lowest bit.
        private int Signed(ref int p)
        {
            int start = p, raw = Compressed(ref p), value = raw >> 1;
            if ((raw & 1) == 0) return value;
            int width = p - start;
            return value - (width == 1 ? 0x40 : width == 2 ? 0x2000 : 0x10000000);
        }

        // The source file, line and column of the instruction at ilOffset in
        // the method (a MethodDef row): the last sequence point at or before it
        // that isn't hidden.
        public bool Locate(int method, int ilOffset, out string file, out int line, out int column)
        {
            file = null;
            line = column = 0;
            if (method > _methodRows) return false;
            int row = _methods + (method - 1) * (_documentIndex + _blobIndex);
            int document = Index(row, _documentIndex), blob = Index(row + _documentIndex, _blobIndex);
            if (blob == 0) return false;
            int p = _blobs + blob, length = Compressed(ref p), end = p + length;
            Compressed(ref p);    // the local signature
            if (document == 0) document = Compressed(ref p);
            int offset = 0, startLine = 0, startColumn = 0, foundDocument = 0;
            bool first = true, anyLine = false;
            while (p < end)
            {
                int delta = Compressed(ref p);
                if (!first && delta == 0) { document = Compressed(ref p); continue; }    // the document changes
                first = false;
                offset += delta;
                int lines = Compressed(ref p);
                int columns = lines == 0 ? Compressed(ref p) : Signed(ref p);
                if (lines == 0 && columns == 0) continue;    // a hidden sequence point
                if (!anyLine) { startLine = Compressed(ref p); startColumn = Compressed(ref p); anyLine = true; }
                else { startLine += Signed(ref p); startColumn += Signed(ref p); }
                if (offset > ilOffset) break;
                line = startLine;
                column = startColumn;
                foundDocument = document;
            }
            if (foundDocument == 0) return false;
            file = DocumentName(foundDocument);
            return file != null;
        }

        // A document's name: a separator, then the blobs of its parts.
        private string DocumentName(int document)
        {
            int p = _blobs + Index(_documents + (document - 1) * _documentRow, _blobIndex);
            int length = Compressed(ref p), end = p + length;
            if (length == 0) return null;
            char separator = (char)_data[p++];
            var name = new StringBuilder();
            for (bool firstPart = true; p < end; firstPart = false)
            {
                if (!firstPart && separator != 0) name.Append(separator);
                int part = _blobs + Compressed(ref p);
                int partLength = Compressed(ref part);
                name.Append(Encoding.UTF8.GetString(_data, part, partLength));
            }
            return name.ToString();
        }
    }

    // ---- The file and its tables --------------------------------------------

    private int U16(int o) => _pe[o] | _pe[o + 1] << 8;
    private int U32(int o) => _pe[o] | _pe[o + 1] << 8 | _pe[o + 2] << 16 | _pe[o + 3] << 24;

    private int CompressedInt(ref int p)
    {
        int length = CompressedLength(p, out int next);
        p = next;
        return length;
    }

    // A compressed unsigned integer at o (II.23.2); start: the byte after it.
    private int CompressedLength(int o, out int start)
    {
        int b = _pe[o];
        if ((b & 0x80) == 0) { start = o + 1; return b; }
        if ((b & 0xC0) == 0x80) { start = o + 2; return (b & 0x3F) << 8 | _pe[o + 1]; }
        start = o + 4;
        return (b & 0x1F) << 24 | _pe[o + 1] << 16 | _pe[o + 2] << 8 | _pe[o + 3];
    }

    private string Str(int offset)
    {
        int start = _strings + offset, end = start;
        while (_pe[end] != 0) end++;
        return Encoding.UTF8.GetString(_pe, start, end - start);
    }

    private int Cell(int table, int row, int column)
    {
        int o = _tableStart[table] + (row - 1) * _rowSize[table] + _columnOffsets[table][column];
        return _columnSizes[table][column] == 4 ? U32(o) : _columnSizes[table][column] == 2 ? U16(o) : _pe[o];
    }

    // The PE headers, the CLI header (returned), the metadata streams, and
    // where each table's rows start.
    private int ReadHeaders()
    {
        int peHeader = U32(0x3C);
        int sectionCount = U16(peHeader + 6);
        int optional = peHeader + 24;
        int dataDirectories = optional + (U16(optional) == 0x20B ? 112 : 96);
        _sections = optional + U16(peHeader + 20);
        _sectionCount = sectionCount;
        int cli = FileOffset(U32(dataDirectories + 14 * 8));
        int root = FileOffset(U32(cli + 8));
        int streamList = root + 16 + U32(root + 12);
        int streamCount = U16(streamList + 2), tables = 0, cursor = streamList + 4;
        for (int i = 0; i < streamCount; i++)
        {
            int offset = root + U32(cursor);
            int nameStart = cursor + 8, nameEnd = nameStart;
            while (_pe[nameEnd] != 0) nameEnd++;
            string name = Encoding.ASCII.GetString(_pe, nameStart, nameEnd - nameStart);
            if (name == "#~") tables = offset;
            else if (name == "#Strings") _strings = offset;
            else if (name == "#Blob") _blobs = offset;
            cursor = nameStart + ((nameEnd - nameStart + 4) & ~3);
        }
        int heapSizes = _pe[tables + 6];
        _stringIndex = (heapSizes & 1) != 0 ? 4 : 2;
        _guidIndex = (heapSizes & 2) != 0 ? 4 : 2;
        _blobIndex = (heapSizes & 4) != 0 ? 4 : 2;
        long valid = (long)(uint)U32(tables + 8) | (long)(uint)U32(tables + 12) << 32;
        int counts = tables + 24;
        for (int t = 0; t < 64; t++)
            if ((valid >> t & 1) != 0) { _rows[t] = U32(counts); counts += 4; }
        int position = counts;
        for (int t = 0; t < Schema.Length; t++)
        {
            object[] columns = Schema[t];
            _columnOffsets[t] = new int[columns.Length];
            _columnSizes[t] = new int[columns.Length];
            int size = 0;
            for (int c = 0; c < columns.Length; c++)
            {
                int width = ColumnSize(columns[c]);
                _columnOffsets[t][c] = size;
                _columnSizes[t][c] = width;
                size += width;
            }
            _rowSize[t] = size;
            _tableStart[t] = position;
            position += _rows[t] * size;
        }
        return cli;
    }

    // Where the file holds a relative virtual address.
    private int FileOffset(int rva)
    {
        for (int i = 0; i < _sectionCount; i++)
        {
            int section = _sections + i * 40, va = U32(section + 12);
            if (rva >= va && rva < va + Math.Max(U32(section + 8), U32(section + 16)))
                return rva - va + U32(section + 20);
        }
        throw new InvalidDataException("An address outside every section of " + Assembly);
    }

    private int ColumnSize(object column)
    {
        if (column is int bytes) return bytes;
        if (column is string heap) return heap == "s" ? _stringIndex : heap == "g" ? _guidIndex : _blobIndex;
        if (column is TableIndex index) return _rows[index.Table] > 0xFFFF ? 4 : 2;
        int[] tables = (int[])column;
        int bits = 0;
        while ((1 << bits) < tables.Length) bits++;
        int max = 0;
        foreach (int t in tables) if (t >= 0) max = Math.Max(max, _rows[t]);
        return max < (1 << (16 - bits)) ? 2 : 4;
    }
